\section{单项选择题(每小题 3 分，共 18 分)}
\begin{enumerate}
    \item \textbf{Solution}. D.

          因为
          \[
              \frac{f(h,0)-f(0,0)}{h}=\frac{|h|}{h}
          \]
          极限不存在，故$f_x'(0,0)$不存在. 又$f(0,y)\equiv0$，所以
          \[
              f_y'(0,0)=\lim_{h\to0}\frac{f(0,h)-f(0,0)}{h}=0
          \]
          存在.

    \item \textbf{Solution}. C.

          在球坐标下，$\sqrt{x^2+y^2+z^2}=\rho$，且$\mathrm{d}v=\rho^2\sin\varphi\,\mathrm{d}\rho\,\mathrm{d}\varphi\,\mathrm{d}\theta$. 区域条件
          \[
              \sqrt{x^2+y^2}\le z\le1
          \]
          化为
          \[
              0\le\varphi\le\frac{\pi}{4},\qquad 0\le\rho\le\frac{1}{\cos\varphi},\qquad 0\le\theta\le2\pi.
          \]
          因此选 C.

    \item \textbf{Solution}. A.

          $\Omega_1$是第一卦限中的单位球部分，关于$x,y,z$轮换对称，故
          \[
              \iiint\limits_{\Omega_1}x\,\mathrm{d}v=\iiint\limits_{\Omega_1}z\,\mathrm{d}v.
          \]
          其余选项中，$\iiint_\Omega z\,\mathrm{d}v>0$，而$\iiint_\Omega xy\,\mathrm{d}v=0$，故不正确.

    \item \textbf{Solution}. C.

          若$\sum(-1)^na_n$收敛，则
          \[
              \sum_{n=1}^{\infty}(a_{2n-1}-a_{2n})
          \]
          正是该交错级数相差一个符号后的分组形式，因此收敛. A 可取$a_n=1/n^2$作反例；B 可取$a_n=(-1)^n/\sqrt n$作反例；D 中比值未取绝对值，一般不能推出收敛.

    \item \textbf{Solution}. B.

          由$\sum a_n$条件收敛知幂级数在$x=2$处收敛但不绝对收敛，因此收敛半径为$1$. 所以$x=3$时$|x-1|=2>1$，级数发散.

    \item \textbf{Solution}. D.

          记
          \[
              P=2xe^{x^2}y^3+ax\cos y,\qquad Q=be^{x^2}y^2-x^2\sin y.
          \]
          路径无关要求$P_y=Q_x$，即
          \[
              6xe^{x^2}y^2-ax\sin y=2bxe^{x^2}y^2-2x\sin y.
          \]
          比较系数得$b=3,\ a=2$.
\end{enumerate}

\begin{enumerate}
    \setcounter{enumi}{6}
    \item \textbf{Solution}. $\mathrm{d}x+2\,\mathrm{d}y+3\,\mathrm{d}z$.

          因为
          \[
              u=\ln x+2\ln y+3\ln z,
          \]
          所以
          \[
              \mathrm{d}u\big|_{(1,1,1)}
              =\frac{1}{x}\,\mathrm{d}x+\frac{2}{y}\,\mathrm{d}y+\frac{3}{z}\,\mathrm{d}z\bigg|_{(1,1,1)}
              =\mathrm{d}x+2\,\mathrm{d}y+3\,\mathrm{d}z.
          \]

    \item \textbf{Solution}. $\dfrac{7}{3}$.

          区域$D$关于$y$轴对称，故$\iint_D x\,\mathrm{d}x\,\mathrm{d}y=0$. 又$D$的面积为$2$，于是
          \[
              \iint_D(1-x)^2\,\mathrm{d}x\,\mathrm{d}y
              =2+\iint_Dx^2\,\mathrm{d}x\,\mathrm{d}y.
          \]
          而
          \[
              \iint_Dx^2\,\mathrm{d}x\,\mathrm{d}y
              =4\int_0^1x^2(1-x)\,\mathrm{d}x=\frac13.
          \]

    \item \textbf{Solution}. $0$.

          当$n\ge1$时，
          \[
              a_n=\frac{1}{\pi}\int_{-\pi}^{\pi}(x+1)\cos nx\,\mathrm{d}x.
          \]
          其中$x\cos nx$为奇函数，积分为$0$；$\int_{-\pi}^{\pi}\cos nx\,\mathrm{d}x=0$，故$a_{2018}=0$.

    \item \textbf{Solution}. $x+y+z$.

          先算
          \[
              \boldsymbol{F}\times\boldsymbol{G}=(xy-z^2,\ yz-x^2,\ xz-y^2),
          \]
          因而
          \[
              \operatorname{div}(\boldsymbol{F}\times\boldsymbol{G})=y+z+x=x+y+z.
          \]
\end{enumerate}

\section{基本计算题(每小题 7 分，共 42 分)}
\begin{enumerate}
    \setcounter{enumi}{10}
    \item \textbf{Solution}. 两曲面的法向量分别为
          \[
              \boldsymbol{n}_1=(2x,2y,2z),\qquad \boldsymbol{n}_2=(2x,2y,-5).
          \]
          在$P(1,2,1)$处，
          \[
              \boldsymbol{n}_1=(2,4,2),\qquad \boldsymbol{n}_2=(2,4,-5).
          \]
          曲线切向量可取作
          \[
              \boldsymbol{\tau}=\boldsymbol{n}_1\times\boldsymbol{n}_2=(-28,14,0)\sim(-2,1,0).
          \]
          故切线方程为
          \[
              \frac{x-1}{-2}=y-2,\qquad z=1.
          \]

    \item \textbf{Solution}. 记
          \[
              s=x+y^2,\qquad t=xy,
          \]
          则$u=f(s,t)$. 先对$x$求偏导：
          \[
              u_x=f_1+yf_2.
          \]
          再对$y$求偏导，得
          \[
              u_{xy}=2yf_{11}+(x+2y^2)f_{12}+xyf_{22}+f_2.
          \]
          其中$f_1,f_2,f_{11},f_{12},f_{22}$均在$(x+y^2,xy)$处取值.

    \item \textbf{Solution}. 对
          \[
              x=r^2\cos\theta,\qquad y=r\sin\theta
          \]
          两边关于$x$求偏导，得
          \[
              \begin{cases}
                  1=2r\cos\theta\,r_x-r^2\sin\theta\,\theta_x,\\
                  0=\sin\theta\,r_x+r\cos\theta\,\theta_x.
              \end{cases}
          \]
          解得
          \[
              \frac{\partial r}{\partial x}=\frac{\cos\theta}{r(1+\cos^2\theta)},\qquad
              \frac{\partial\theta}{\partial x}=-\frac{\sin\theta}{r^2(1+\cos^2\theta)}.
          \]

    \item \textbf{Solution}. $L$由$x$轴线段、$y$轴线段及第一象限单位圆弧组成. 在$x$轴上，
          \[
              I_1=\int_0^1x\,\mathrm{d}x=\frac12.
          \]
          在$y$轴上，
          \[
              I_2=\int_0^1y^2\,\mathrm{d}y=\frac13.
          \]
          在圆弧$x=\cos t,\ y=\sin t\ (0\le t\le\pi/2)$上，$\mathrm{d}s=\mathrm{d}t$，故
          \[
              I_3=\int_0^{\pi/2}(\cos t+\sin^2t)\,\mathrm{d}t=1+\frac{\pi}{4}.
          \]
          所以
          \[
              I=I_1+I_2+I_3=\frac{11}{6}+\frac{\pi}{4}.
          \]

    \item \textbf{Solution}. 设
          \[
              P=xy^2,\qquad Q=yz^2,\qquad R=zx^2.
          \]
          曲面为$z=\sqrt{x^2+y^2}$，取上侧，则
          \[
              \mathrm{d}y\,\mathrm{d}z=-z_x\,\mathrm{d}x\,\mathrm{d}y,\qquad
              \mathrm{d}z\,\mathrm{d}x=-z_y\,\mathrm{d}x\,\mathrm{d}y.
          \]
          令$x=r\cos\theta,\ y=r\sin\theta,\ z=r$，投影区域为$0\le r\le1$，于是
          \[
              \begin{aligned}
                  I
                  &=\iint_D(-Pz_x-Qz_y+R)\,\mathrm{d}x\,\mathrm{d}y\\
                  &=\int_0^{2\pi}\int_0^1r^4(\cos^2\theta-\sin^2\theta-\cos^2\theta\sin^2\theta)\,\mathrm{d}r\,\mathrm{d}\theta\\
                  &=-\frac{\pi}{20}.
              \end{aligned}
          \]

    \item \textbf{Solution}. 由
          \[
              \arctan x=\sum_{n=0}^{\infty}(-1)^n\frac{x^{2n+1}}{2n+1}\qquad(|x|\le1)
          \]
          可知
          \[
              \sum_{n=0}^{\infty}(-1)^n\frac{x^{2n}}{2n+1}
              =
              \begin{cases}
                  \dfrac{\arctan x}{x}, & x\ne0,\\[0.6em]
                  1,                    & x=0.
              \end{cases}
          \]
          收敛域为$[-1,1]$. 取$x=\dfrac12$，得
          \[
              \sum_{n=0}^{\infty}\frac{(-1)^n}{4^n(2n+1)}
              =2\arctan\frac12.
          \]
\end{enumerate}

\section{应用题(每小题 7 分，共 14 分)}
\begin{enumerate}
    \setcounter{enumi}{16}
    \item \textbf{Solution}. 在第一卦限内令
          \[
              \Phi(x,y,z)=\ln x+2\ln y+3\ln z.
          \]
          由 Lagrange 乘数法，
          \[
              \frac1x=2\lambda x,\qquad \frac2y=4\lambda y,\qquad \frac3z=6\lambda z,
          \]
          所以
          \[
              x^2=y^2=z^2.
          \]
          结合$x^2+2y^2+3z^2=6R^2$，且$x,y,z>0$，得最大值点为
          \[
              (R,R,R),
          \]
          最大值为
          \[
              \ln(R\cdot R^2\cdot R^3)=\ln R^6.
          \]
          对任意正实数$a,b,c$，取
          \[
              R^2=\frac{a+2b+3c}{6},\qquad x=\sqrt a,\quad y=\sqrt b,\quad z=\sqrt c.
          \]
          则点$(x,y,z)$在椭球面上，故
          \[
              \ln\sqrt{ab^2c^3}\le \ln R^6.
          \]
          因而
          \[
              ab^2c^3\le R^{12}=\left(\frac{a+2b+3c}{6}\right)^6.
          \]

    \item \textbf{Solution}. 用柱坐标. 柱面$x^2+y^2=2x$化为
          \[
              r=2\cos\theta,\qquad -\frac{\pi}{2}\le\theta\le\frac{\pi}{2}.
          \]
          区域为
          \[
              -\frac{\pi}{2}\le\theta\le\frac{\pi}{2},\qquad 0\le r\le2\cos\theta,\qquad 0\le z\le r.
          \]
          因而
          \[
              V=\int_{-\pi/2}^{\pi/2}\int_0^{2\cos\theta}\int_0^r r\,\mathrm{d}z\,\mathrm{d}r\,\mathrm{d}\theta
              =\frac{32}{9}.
          \]
          锥面$z=\sqrt{x^2+y^2}$的面积元为
          \[
              \mathrm{d}S=\sqrt2\,\mathrm{d}x\,\mathrm{d}y.
          \]
          其投影为圆盘$x^2+y^2\le2x$，面积为$\pi$，故
          \[
              S=\sqrt2\,\pi.
          \]
\end{enumerate}

\section{分析证明题(每小题 5 分，共 10 分)}
\begin{enumerate}
    \setcounter{enumi}{18}
    \item \textbf{Solution}. 记
          \[
              P=-\frac{y}{4x^2+y^2},\qquad Q=\frac{x}{4x^2+y^2}.
          \]
          在不含原点的区域内，有
          \[
              Q_x-P_y=0.
          \]
          当$0<R<1$时，圆周$L$内部不含奇点$(0,0)$，由 Green 公式得
          \[
              I=0.
          \]
          当$R>1$时，$L$包围原点. 令$u=2x,\ v=y$，则
          \[
              \frac{x\,\mathrm{d}y-y\,\mathrm{d}x}{4x^2+y^2}
              =\frac12\frac{u\,\mathrm{d}v-v\,\mathrm{d}u}{u^2+v^2}.
          \]
          因线性变换保持正向且曲线绕原点一周，
          \[
              I=\frac12\cdot2\pi=\pi.
          \]
          综上，
          \[
              I=
              \begin{cases}
                  0,  & 0<R<1,\\
                  \pi,& R>1.
              \end{cases}
          \]

    \item \textbf{Proof}. 由 Cauchy-Schwarz 不等式，
          \[
              \left(\int_0^1e^{f(x)}\,\mathrm{d}x\right)
              \left(\int_0^1e^{-f(x)}\,\mathrm{d}x\right)
              \ge
              \left(\int_0^1e^{f(x)/2}e^{-f(x)/2}\,\mathrm{d}x\right)^2.
          \]
          右端为
          \[
              \left(\int_0^1 1\,\mathrm{d}x\right)^2=1,
          \]
          因而原不等式成立.
\end{enumerate}
