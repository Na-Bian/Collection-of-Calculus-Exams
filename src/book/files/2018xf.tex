          由题意可知$f(x,0)=|x|$，因此
          \[
              \lim_{h\to0}\frac{f(h,0)-f(0,0)}{h}=\lim_{h\to0}\frac{|h|}{h}
          \]
          极限不存在，即$f_x'(0,0)$不存在. 又$f(0,y)\equiv0$，所以
          \[
              f_y'(0,0)=\lim_{h\to0}\frac{f(0,h)-f(0,0)}{h}=0
          \]
          存在.

    \item \textbf{Solution}. C.

          在球坐标下，$\sqrt{x^2+y^2+z^2}=\rho$，且$\mathrm{d}v=\rho^2\sin\varphi\,\mathrm{d}\rho\,\mathrm{d}\varphi\,\mathrm{d}\theta$. 区域条件
          \[
              \sqrt{x^2+y^2}\le z\le1
          \]
          化为
          \[
              0\le\varphi\le\frac{\pi}{4},\qquad 0\le\rho\le\frac{1}{\cos\varphi},\qquad 0\le\theta\le2\pi.
          \]
          因此$I=\int_{0}^{2\pi}\,\mathrm{d}\theta\int_0^{\frac{\pi}{4}}\sin\varphi\,\mathrm{d}\varphi\int_0^{\frac{1}{\cos\varphi}}f(\rho)\rho^2\,\mathrm{d}\rho=2\pi\int_0^{\frac{\pi}{4}}\sin\varphi\,\mathrm{d}\varphi\int_0^{\frac{1}{\cos\varphi}}f(\rho)\rho^2\,\mathrm{d}\rho$.

          $\Omega_1$是第一卦限中的单位球部分，关于$x,y,z$轮换对称，故
          \[
              \iiint\limits_{\Omega_1}x\,\mathrm{d}v=\iiint\limits_{\Omega_1}z\,\mathrm{d}v.
          \]
          而由对称性可知$\iiint\limits_\Omega z\,\mathrm{d}v>0$，$\iiint\limits_\Omega xy\,\mathrm{d}v=0$.

          \]

              \end{cases}
          \]
          解得最大值点为$(R,R,R)$，最大值为
          \[
              \ln(R\cdot R^2\cdot R^3)=6\ln R.
          \]
          对任意正实数$a,b,c$，取
          \[
              R^2=\frac{a+2b+3c}{6},\qquad x=\sqrt a,\quad y=\sqrt b,\quad z=\sqrt c.
          \]
          则点$(x,y,z)$在椭球面上，故
          \[
              \ln\sqrt{ab^2c^3}\le 6\ln R.
          \]
          因而
          \[
              ab^2c^3\le R^{12}=\left(\frac{a+2b+3c}{6}\right)^6.
          \]

    \item \textbf{Solution}. 记
          \[
              P=-\frac{y}{4x^2+y^2},\qquad Q=\frac{x}{4x^2+y^2}.
          \]
          在不含原点的区域内，有
          \[
              Q_x-P_y=0.
          \]
          当$0<R<1$时，圆周$L$内部不含奇点$(0,0)$，由 Green 公式得
          \[
              I=0.
          \]
          当$R>1$时，$L$包围原点. 考虑充分小的$\delta>0$，使得小椭圆$E:4x^2+y^2=\delta^2$包含在$L$内部.

          从而由复合闭路定理，
          \[
              I=\oint_L P\,\mathrm{d}x+Q\,\mathrm{d}y=\oint_E \frac{x\,\mathrm{d}y-y\,\mathrm{d}x}{\delta^2}.
          \]
          由Green公式，$\oint_E x\,\mathrm{d}y-y\,\mathrm{d}x=2\iint\limits_{4x^2+y^2\le\delta^2}\,\mathrm{d}x\,\mathrm{d}y=\pi\delta^2$. 因此$I=\pi$.

          综上所述，
          \[
              I=\begin{cases}
                  0,   & 0<R<1, \\
                  \pi, & R>1.
              \end{cases}
          \]

    \item \textbf{Proof}. 记$D:0\le x\le1,0\le y\le1$，$I=\int_0^1\mathrm{e}^{f(x)}\,\mathrm{d}x\int_0^1\mathrm{e}^{-f(x)}\,\mathrm{d}x$，则
          \[
              I=\int_0^1\mathrm{e}^{f(x)}\,\mathrm{d}x\int_0^1\mathrm{e}^{-f(y)}\,\mathrm{d}y
              =\iint\limits_D\mathrm{e}^{f(x)-f(y)}\,\mathrm{d}x\,\mathrm{d}y.
          \]
          由$D$的轮换对称性可知$I=\iint\limits_D\mathrm{e}^{f(y)-f(x)}\,\mathrm{d}x\,\mathrm{d}y$，因此
          \[
              \begin{aligned}
                  I={}   & \frac{1}{2}\iint\limits_D\left(\mathrm{e}^{f(x)-f(y)}+\mathrm{e}^{f(y)-f(x)}\right)\,\mathrm{d}x\,\mathrm{d}y           \\
                  ={}    & \frac{1}{2}\iint\limits_D\left(\mathrm{e}^{f(x)-f(y)}+\frac{1}{\mathrm{e}^{f(x)-f(y)}}\right)\,\mathrm{d}x\,\mathrm{d}y \\
                  \geq{} & \iint\limits_D\,\mathrm{d}x\,\mathrm{d}y=1.
              \end{aligned}
          \]

\end{enumerate}
