      \[
        \begin{aligned}
          1 = \lim_{x\to1}\frac{\ln x^2}{c(x-1)^k}= & \lim_{x\to1}\frac{2\ln x}{c(x-1)^k}                                             \\
          =                                         & \lim_{x\to1}\frac{2}{c k (x-1)^{k-1} x}= \lim_{x\to1}\frac{2}{c k (x-1)^{k-1}}.
        \end{aligned}
      \]

      所以$k=1$，$c=2$，无穷小主部为$2(x-1)$.

      若$f(0)=0$，因为$f(x)$在$x=0$处连续，所以$\lim_{x\to0}f(x)=0$.

      又$F(0)=0$，则
      \[
        \begin{aligned}
          \lim_{x\to0}\frac{F(x)-F(0)}{x-0}= & \lim_{x\to0}\frac{f(x)|\sin x|}{x}       \\
          =                                  & \lim_{x\to0}f(x)\cdot\frac{|\sin x|}{x}.
        \end{aligned}
      \]

      由于$\lim_{x\to0}f(x)=0$，且$\frac{|\sin x|}{x}$在$x\to0$时是有界变量，无穷小量乘有界变量仍为无穷小量，

      故$F'(0)=0$，即$F'(0)$存在.

      若$F(x)$在$x=0$处可导，则$\lim_{x\to0}\frac{F(x)-F(0)}{x}$存在，即$\lim_{x\to0}
      \frac{f(x)|\sin x|}{x}$存在.

      分别考察左右极限：
      \[
        \lim_{x\to0^+}\frac{f(x)|\sin x|}{x}=\lim_{x\to0^+}f(x)\frac{\sin x}{x}=f
        (0)\cdot 1 = f(0),
      \]
      \[
        \lim_{x\to0^-}\frac{f(x)|\sin x|}{x}=\lim_{x\to0^-}f(x)\frac{-\sin x}{x}=
        f(0)\cdot (-1) = -f(0).
      \]

      因为极限存在，左右极限必须相等，所以$f(0)=-f(0)$，解得$f(0)=0$.

      因此$f(0)=0$是$F'(0)$存在的充要条件.

  \begin{enumerate}

    \item \textbf{Proof}. 由题意可知，存在$c\in(a,b)$，使得$f'(c)=0$且$f(c)=
      -1$.

      由Taylor公式，
      \[
        \begin{gathered}
          f(a)-f(c)=2=f'(c)(a-c)+\frac{f''(\xi_{1})}{2}(a-c)^2=\frac{f''(\xi_{1})}{2}(c-a)^2,
          \\
          f(b)-f(c)=2=f'(c)(b-c)+\frac{f''(\xi_{2})}{2}(b-c)^2=\frac{f''(\xi_{2})}{2}(b-c)^2,
        \end{gathered}
      \]

      其中$\xi_{1},\xi_{2}$介于$a$与$b$之间. 上述两式即$f''(\xi_{1})=\frac{16}{[2(c-a)]^{2}}$，$f
      ''(\xi_{2})=\frac{16}{[2(b-c)]^{2}}$.

      记$M=\max\{2(c-a),2(b-c)\}$，$N=\min\{2(c-a),2(b-c)\}$，则$N\leq b-a \leq M$，

      所以$\frac{16}{M^{2}}\leq\frac{16}{(b-a)^{2}}\leq\frac{16}{N^{2}}$.

      因为$f''(x)$在$[a,b]$上连续，由介值定理可知，存在$\xi\in(a,b)$使得$f''(\xi)
      =\frac{16}{(b-a)^{2}}$.

    \item \textbf{Solution}. 令$F(t)=\int_{a}^{t}(x-a)^{3} f(x)\,\mathrm{d}x-\frac{(t-a)^{3}}{4}
      \int_{a}^{t}f(x)\,\mathrm{d}x$（$a\leq t \leq b$），

      则
      \[
        \begin{aligned}
          F'(t) = & (t-a)^{3} f(t) - \frac{3(t-a)^2}{4}\int_{a}^{t}f(x)\,\mathrm{d}x - \frac{(t-a)^3}{4}f(t) \\
          =       & \frac{3(t-a)^3}{4}f(t) - \frac{3(t-a)^2}{4}\int_{a}^{t}f(x)\,\mathrm{d}x                 \\
          =       & \frac{3(t-a)^2}{4}\left[(t-a)f(t) - \int_{a}^{t}f(x)\,\mathrm{d}x\right].
        \end{aligned}
      \]

      由积分中值定理，存在$\xi\in(a,t)$使得$\int_{a}^{t}f(x)\,\mathrm{d}x = (t-a)f(
      \xi)$，

      所以$F'(t) = \frac{3(t-a)^{2}}{4}\left[(t-a)f(t) - \int_{a}^{t}f(x)\,\mathrm{d}
      x\right] = \frac{3(t-a)^{3}}{4}[f(t) - f(\xi)]$.

      因为$f(x)$在$[a,b]$上单调递减，所以$f(t)\leq f(\xi)$，故$F'(t)\leq0$，$F(t)$在$[
      a,b]$上单调递减.

      因此$F(b) \leq F(a) = 0$，即$\int_{a}^{b}(x-a)^{3} f(x)\,\mathrm{d}x \leq \frac{(b-a)^{3}}{4}
      \int_{a}^{b}f(x)\,\mathrm{d}x$.
  \end{enumerate}
