                  \frac{\partial F}{\partial y}=2(y+1)+\lambda(x+2y)=0, \\[6pt]
                  \frac{\partial F}{\partial \lambda}=x^2+xy+y^2-3=0,
              \end{cases}
          \]

          解得$\begin{cases}
                  x=1, \\
                  y=1,
              \end{cases}$，$\begin{cases}
                  x=-1, \\
                  y=-1,
              \end{cases}$，$\begin{cases}
                  x=-1, \\
                  y=2,
              \end{cases}$，或$\begin{cases}
                  x=2, \\
                  y=-1.
              \end{cases}$

          计算$d(1,1)=2\sqrt{2}$，$d(-1,-1)=0$，$d(2,-1)=3$，$d(-1,2)=3$，所以最大值为$3$.

\end{enumerate}

    \item \textbf{Proof}. 由轮换对称性，
          \[
              \iint\limits_D \cos y^2\,\mathrm{d}x\,\mathrm{d}y=\iint\limits_D \cos x^2\,\mathrm{d}x\,\mathrm{d}y.

          所以
          \[

              \frac{\sqrt2}{2}\le \sin\left(x^2+\frac{\pi}{4}\right)\le 1.

              \begin{gathered}

              \end{gathered}

    \item \textbf{Proof}. $a_n=\frac{(-1)^n}{\sqrt{n+(-1)^n}}=\frac{(-1)^n}{\sqrt{n}}\left[1+\frac{(-1)^n}{n}\right]^{-\frac{1}{2}}$，
